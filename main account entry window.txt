//main account entry window

// main_account_window.dart
import 'package:flutter/material.dart';
import 'entering_acc_names_window.dart';
import 'home_window.dart';

class MainAccountWindow extends StatefulWidget {
  const MainAccountWindow({super.key});

  @override
  State<MainAccountWindow> createState() => _MainAccountWindowState();
}

class _MainAccountWindowState extends State<MainAccountWindow> {
  String? selectedAccountType;
  String? selectedMainAccount;
  final TextEditingController accountNumberController = TextEditingController();
  final TextEditingController mainAccountSearchController = TextEditingController();
  final TextEditingController secondaryFieldController = TextEditingController();
  bool saveInTalebandh = false;
  bool isJama = true;
  bool _isAccountNumberUnique = true;
  MainAccountEntry? selectedSavedEntry;

  // Updated account types list
  final List<String> accountTypes = [
    'देणेकरी',
    'शिल्लक माल',
    'नफा तोटा (जमा)',
    'नफा तोटा (नावे)',
    'रोख व बँक शिल्लक',
    'ताळेबंद (इतर)', // This will be shown as "ताळेबंद" when checkbox is checked
    'व्यापारी(जमा)',
    'व्यापारी(नावे)',
    'येणेकरी'
  ];

  @override
  void initState() {
    super.initState();
    _loadExistingEntries();
  }

  void _loadExistingEntries() {
    // The entries are already in the repository, no need to load separately
  }

  List<String> get mainAccounts {
    final repo = DairyDataRepo.instance;
    return repo.mainAccountNames;
  }

  List<String> get filteredMainAccounts {
    if (mainAccountSearchController.text.isEmpty) {
      return mainAccounts;
    }
    return mainAccounts
        .where((account) => account.toLowerCase().contains(mainAccountSearchController.text.toLowerCase()))
        .toList();
  }

  void _validateAccountNumber(String value) {
    final repo = DairyDataRepo.instance;
    final isUnique = !repo.mainAccountEntries.value.any((entry) =>
    entry.accountNumber == value &&
        (selectedSavedEntry == null || entry.accountNumber != selectedSavedEntry!.accountNumber));

    setState(() {
      _isAccountNumberUnique = isUnique;
    });
  }

  Widget _buildAccountNumberField() {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        const Text(
          'अनु. नं.',
          style: TextStyle(
            fontWeight: FontWeight.w600,
            fontSize: 14,
            color: Color(0xFF1E3C72),
          ),
        ),
        const SizedBox(height: 8),
        Container(
          height: 48,
          decoration: BoxDecoration(
            color: Colors.white,
            borderRadius: BorderRadius.circular(8),
            border: Border.all(
              color: _isAccountNumberUnique ? Colors.grey.shade300 : Colors.red,
              width: _isAccountNumberUnique ? 1 : 2,
            ),
          ),
          child: TextField(
            controller: accountNumberController,
            decoration: InputDecoration(
              hintText: 'अनु. नं. प्रविष्ट करा',
              contentPadding: const EdgeInsets.symmetric(horizontal: 12, vertical: 12),
              border: InputBorder.none,
              hintStyle: const TextStyle(
                color: Color(0xFF666666),
                fontSize: 14,
              ),
              errorText: _isAccountNumberUnique ? null : 'हा अनु. नं. आधीच वापरात आहे',
              errorStyle: const TextStyle(
                color: Colors.red,
                fontSize: 12,
              ),
            ),
            style: const TextStyle(
              fontSize: 14,
              fontWeight: FontWeight.w500,
            ),
            onChanged: (value) {
              _validateAccountNumber(value);
            },
          ),
        ),
        if (!_isAccountNumberUnique)
          const Padding(
            padding: EdgeInsets.only(top: 4),
            child: Text(
              'कृपया वेगळा अनु. नं. प्रविष्ट करा',
              style: TextStyle(
                color: Colors.red,
                fontSize: 12,
              ),
            ),
          ),
      ],
    );
  }

  // Method to get display name for account type based on checkbox state
  String _getAccountTypeDisplayName(String accountType) {
    if (accountType == 'ताळेबंद (इतर)' && saveInTalebandh) {
      return 'ताळेबंद';
    }
    return accountType;
  }

  void _loadSavedEntry(MainAccountEntry? entry) {
    if (entry != null) {
      setState(() {
        selectedSavedEntry = entry;
        accountNumberController.text = entry.accountNumber;
        selectedAccountType = entry.accountType;
        selectedMainAccount = entry.mainAccount;
        saveInTalebandh = entry.showInBalance;
        mainAccountSearchController.text = entry.mainAccount;
        _isAccountNumberUnique = true;
      });
    } else {
      _clearForm();
    }
  }

  void _clearForm() {
    setState(() {
      selectedSavedEntry = null;
      accountNumberController.clear();
      selectedAccountType = null;
      selectedMainAccount = null;
      saveInTalebandh = false;
      mainAccountSearchController.clear();
      secondaryFieldController.clear();
      isJama = true;
      _isAccountNumberUnique = true;
    });
  }

  void _saveEntry() {
    if (selectedAccountType == null ||
        mainAccountSearchController.text.isEmpty ||
        accountNumberController.text.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text("कृपया सर्व आवश्यक फील्ड भरा")),
      );
      return;
    }

    // Check if account number is unique
    if (!_isAccountNumberUnique) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text("कृपया वेगळा अनु. नं. प्रविष्ट करा")),
      );
      return;
    }

    final newEntry = MainAccountEntry(
      accountNumber: accountNumberController.text,
      accountType: selectedAccountType!,
      mainAccount: mainAccountSearchController.text,
      showInBalance: saveInTalebandh,
    );

    final repo = DairyDataRepo.instance;

    // If editing existing entry, update it
    if (selectedSavedEntry != null) {
      repo.updateMainAccountEntry(selectedSavedEntry!, newEntry);
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text("खाते यशस्वीरित्या अपडेट केले")),
      );
    } else {
      // Add new entry to repository
      repo.addMainAccountEntry(newEntry);
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text("खाते यशस्वीरित्या जतन केले")),
      );
    }

    _clearForm();
  }

  void _deleteEntry() {
    if (selectedSavedEntry == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text("कृपया हटवण्यासाठी खाते निवडा")),
      );
      return;
    }

    final repo = DairyDataRepo.instance;
    repo.deleteMainAccountEntry(selectedSavedEntry!);

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text("खाते यशस्वीरित्या हटवले: ${selectedSavedEntry!.accountNumber}")),
    );

    _clearForm();
  }

  void _updateEntry() {
    if (selectedSavedEntry == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text("कृपया अपडेट करण्यासाठी खाते निवडा")),
      );
      return;
    }

    _saveEntry();
  }

  Widget _buildHeader() {
    return Container(
      decoration: BoxDecoration(
        gradient: const LinearGradient(
          colors: [Color(0xFF1E3C72), Color(0xFF2A5298)],
          begin: Alignment.topLeft,
          end: Alignment.bottomRight,
        ),
        borderRadius: const BorderRadius.only(
          topLeft: Radius.circular(16),
          topRight: Radius.circular(16),
        ),
        boxShadow: [
          BoxShadow(
            color: const Color(0xFF1E3C72).withOpacity(0.4),
            blurRadius: 8,
            offset: const Offset(0, 3),
          ),
        ],
      ),
      padding: const EdgeInsets.symmetric(vertical: 20),
      child: Center(
        child: Row(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Container(
              padding: const EdgeInsets.all(10),
              decoration: BoxDecoration(
                color: Colors.white.withOpacity(0.15),
                shape: BoxShape.circle,
              ),
              child: const Icon(
                Icons.account_balance_rounded,
                color: Colors.white,
                size: 26,
              ),
            ),
            const SizedBox(width: 16),
            const Text(
              "मुख्य खाते नावे भरणे",
              style: TextStyle(
                color: Colors.white,
                fontSize: 22,
                fontWeight: FontWeight.w800,
                letterSpacing: 0.6,
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _buildPremiumDropdown(String label, String? value, List<String> items, Function(String?) onChanged) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(
          label,
          style: const TextStyle(
            fontWeight: FontWeight.w600,
            fontSize: 14,
            color: Color(0xFF1E3C72),
          ),
        ),
        const SizedBox(height: 8),
        Container(
          padding: const EdgeInsets.symmetric(horizontal: 12),
          decoration: BoxDecoration(
            color: Colors.white,
            borderRadius: BorderRadius.circular(8),
            border: Border.all(color: Colors.grey.shade300),
          ),
          child: DropdownButton<String>(
            value: value,
            isExpanded: true,
            underline: const SizedBox.shrink(),
            icon: const Icon(Icons.arrow_drop_down_rounded, color: Color(0xFF1E6BC9), size: 20),
            style: const TextStyle(
              fontSize: 14,
              fontWeight: FontWeight.w500,
              color: Colors.black87,
            ),
            hint: Text(
              '$label निवडा',
              style: const TextStyle(
                color: Color(0xFF666666),
                fontSize: 14,
              ),
            ),
            items: items
                .map((item) => DropdownMenuItem(
              value: item,
              child: Text(_getAccountTypeDisplayName(item)), // Use display name here
            ))
                .toList(),
            onChanged: onChanged,
          ),
        ),
      ],
    );
  }

  Widget _buildSearchableMainAccountField() {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        const Text(
          'मुख्य खाते',
          style: TextStyle(
            fontWeight: FontWeight.w600,
            fontSize: 14,
            color: Color(0xFF1E3C72),
          ),
        ),
        const SizedBox(height: 8),
        Container(
          decoration: BoxDecoration(
            color: Colors.white,
            borderRadius: BorderRadius.circular(8),
            border: Border.all(color: Colors.grey.shade300),
          ),
          child: Row(
            children: [
              Expanded(
                child: TextField(
                  controller: mainAccountSearchController,
                  decoration: InputDecoration(
                    hintText: DairyDataRepo.instance.mainAccountEntries.value.isNotEmpty
                        ? 'जुने खाते निवडा किंवा नवीन खाते नाव टाइप करा'
                        : 'नवीन खाते नाव टाइप करा',
                    contentPadding: const EdgeInsets.symmetric(horizontal: 12, vertical: 12),
                    border: InputBorder.none,
                    hintStyle: const TextStyle(
                      color: Color(0xFF666666),
                      fontSize: 14,
                    ),
                  ),
                  style: const TextStyle(
                    fontSize: 14,
                    fontWeight: FontWeight.w500,
                  ),
                  onChanged: (value) {
                    setState(() {
                      selectedMainAccount = value;
                      if (value.isNotEmpty) {
                        selectedSavedEntry = null;
                      }
                    });
                  },
                ),
              ),
              ValueListenableBuilder<List<MainAccountEntry>>(
                valueListenable: DairyDataRepo.instance.mainAccountEntries,
                builder: (context, savedEntries, _) {
                  if (savedEntries.isEmpty) {
                    return const SizedBox.shrink();
                  }
                  return PopupMenuButton<MainAccountEntry>(
                    icon: const Icon(Icons.arrow_drop_down_rounded, color: Color(0xFF1E6BC9)),
                    onSelected: (entry) {
                      _loadSavedEntry(entry);
                    },
                    itemBuilder: (context) {
                      return savedEntries
                          .map((entry) => PopupMenuItem(
                        value: entry,
                        child: Text(
                          entry.toString(),
                          style: const TextStyle(fontSize: 14),
                        ),
                      ))
                          .toList();
                    },
                  );
                },
              ),
            ],
          ),
        ),
        const SizedBox(height: 4),
        Text(
          DairyDataRepo.instance.mainAccountEntries.value.isEmpty
              ? 'टीप: नवीन खाते नाव टाइप करा'
              : 'टीप: ड्रॉपडाउन आयकॉन वर क्लिक करून जुने खाते निवडा किंवा नवीन टाइप करा',
          style: TextStyle(
            fontSize: 12,
            color: Colors.grey.shade600,
            fontStyle: FontStyle.italic,
          ),
        ),
      ],
    );
  }

  Widget _buildPremiumButton({
    required IconData icon,
    required String label,
    required Color backgroundColor,
    required Color textColor,
    Color? borderColor,
    required VoidCallback onPressed,
  }) {
    return ElevatedButton.icon(
      icon: Icon(icon, size: 18, color: textColor),
      label: Text(
        label,
        style: TextStyle(
          fontSize: 13,
          fontWeight: FontWeight.w600,
          color: textColor,
        ),
      ),
      style: ElevatedButton.styleFrom(
        backgroundColor: backgroundColor,
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.circular(8),
          side: borderColor != null
              ? BorderSide(color: borderColor, width: 1)
              : BorderSide.none,
        ),
        padding: const EdgeInsets.symmetric(horizontal: 20, vertical: 12),
        elevation: 0,
      ),
      onPressed: onPressed,
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFFE9F0FB),
      body: Center(
        child: Container(
          width: 700,
          height: 600,
          decoration: BoxDecoration(
            color: Colors.white,
            borderRadius: BorderRadius.circular(16),
            boxShadow: const [
              BoxShadow(
                color: Colors.black26,
                blurRadius: 10,
                offset: Offset(2, 3),
              )
            ],
          ),
          child: Column(
            children: [
              _buildHeader(),

              // Main Content
              Expanded(
                child: SingleChildScrollView(
                  padding: const EdgeInsets.all(24.0),
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [

                      // Account Type Dropdown
                      _buildPremiumDropdown(
                        'खाते प्रकार',
                        selectedAccountType,
                        accountTypes,
                            (value) {
                          setState(() => selectedAccountType = value);
                        },
                      ),
                      const SizedBox(height: 20),

                      // Searchable Main Account Field
                      _buildSearchableMainAccountField(),
                      const SizedBox(height: 20),

                      // Account Number Input and Checkbox
                      Row(
                        crossAxisAlignment: CrossAxisAlignment.start,
                        children: [
                          // Account Number Column
                          Expanded(
                            child: _buildAccountNumberField(),
                          ),
                          const SizedBox(width: 16),

                          // Checkbox Column
                          Expanded(
                            child: Column(
                              crossAxisAlignment: CrossAxisAlignment.start,
                              children: [
                                const SizedBox(height: 25),
                                Container(
                                  height: 48,
                                  child: Row(
                                    children: [
                                      Checkbox(
                                        value: saveInTalebandh,
                                        onChanged: (val) {
                                          setState(() => saveInTalebandh = val ?? false);
                                        },
                                        activeColor: const Color(0xFF1E3C72),
                                        materialTapTargetSize: MaterialTapTargetSize.shrinkWrap,
                                        visualDensity: VisualDensity.compact,
                                      ),
                                      const SizedBox(width: 4),
                                      const Expanded(
                                        child: Text(
                                          'ताळेबंद खातं दाखवा',
                                          style: TextStyle(
                                            fontSize: 13,
                                            fontWeight: FontWeight.w600,
                                            color: Color(0xFF1E3C72),
                                          ),
                                        ),
                                      ),
                                    ],
                                  ),
                                ),
                              ],
                            ),
                          ),
                        ],
                      ),
                      const SizedBox(height: 30),

                      // Action Buttons
                      Center(
                        child: Wrap(
                          spacing: 12,
                          runSpacing: 12,
                          alignment: WrapAlignment.center,
                          children: [
                            _buildPremiumButton(
                              icon: Icons.note_add,
                              label: 'नवीन',
                              backgroundColor: const Color(0xFF1E3C72).withOpacity(0.1),
                              textColor: const Color(0xFF1E3C72),
                              onPressed: _clearForm,
                            ),

                            _buildPremiumButton(
                              icon: Icons.check_circle,
                              label: 'जतन करा',
                              backgroundColor: const Color(0xFF1E3C72),
                              textColor: Colors.white,
                              onPressed: _saveEntry,
                            ),

                            _buildPremiumButton(
                              icon: Icons.update,
                              label: 'अपडेट',
                              backgroundColor: Colors.orange.shade600,
                              textColor: Colors.white,
                              onPressed: _updateEntry,
                            ),

                            _buildPremiumButton(
                              icon: Icons.delete,
                              label: 'हटवा',
                              backgroundColor: Colors.red.shade700,
                              textColor: Colors.white,
                              onPressed: _deleteEntry,
                            ),

                            _buildPremiumButton(
                              icon: Icons.stop_circle,
                              label: 'थांबवा',
                              backgroundColor: Colors.grey.shade600,
                              textColor: Colors.white,
                              onPressed: () {
                                Navigator.of(context).pushReplacement(
                                  MaterialPageRoute(builder: (context) => HomeScreen()),
                                );
                              },
                            ),
                          ],
                        ),
                      ),
                    ],
                  ),
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
